import javax.swing.*;
import java.awt.*;
import java.awt.event.ActionEvent;
import java.awt.event.ActionListener;

/**
 * Question 2: Water Tank Surface Area Calculator GUI Application
 * Course: Computer Literacy (HCLT107-1)
 */
public class Question2 extends JFrame implements ActionListener {
    private JTextField txtRadius, txtHeight;
    private JTextArea txtResult;
    private JButton btnCalculate, btnClear, btnExit;

    public Question2() {
        setTitle("Question 2 - Water Tank Surface Area Calculator");
        setSize(450, 380);
        setDefaultCloseOperation(JFrame.DISPOSE_ON_CLOSE);
        setLocationRelativeTo(null);
        setLayout(new BorderLayout(10, 10));

        // Input Panel
        JPanel inputPanel = new JPanel(new GridLayout(2, 2, 5, 5));
        inputPanel.setBorder(BorderFactory.createEmptyBorder(10, 10, 10, 10));
        inputPanel.add(new JLabel("Radius (r):"));
        txtRadius = new JTextField();
        inputPanel.add(txtRadius);
        inputPanel.add(new JLabel("Height (h):"));
        txtHeight = new JTextField();
        inputPanel.add(txtHeight);
        add(inputPanel, BorderLayout.NORTH);

        // Result Display Area
        txtResult = new JTextArea();
        txtResult.setEditable(false);
        txtResult.setLineWrap(true);
        txtResult.setWrapStyleWord(true);
        JScrollPane scrollPane = new JScrollPane(txtResult);
        add(scrollPane, BorderLayout.CENTER);

        // Control Buttons
        JPanel buttonPanel = new JPanel(new FlowLayout());
        btnCalculate = new JButton("Calculate");
        btnClear = new JButton("Clear");
        btnExit = new JButton("Exit");

        btnCalculate.addActionListener(this);
        btnClear.addActionListener(this);
        btnExit.addActionListener(this);

        buttonPanel.add(btnCalculate);
        buttonPanel.add(btnClear);
        buttonPanel.add(btnExit);
        add(buttonPanel, BorderLayout.SOUTH);
    }

    @Override
    public void actionPerformed(ActionEvent e) {
        if (e.getSource() == btnCalculate) {
            calculateSurfaceArea();
        } else if (e.getSource() == btnClear) {
            txtRadius.setText("");
            txtHeight.setText("");
            txtResult.setText("");
            txtRadius.requestFocus();
        } else if (e.getSource() == btnExit) {
            this.dispose();
        }
    }

    private void calculateSurfaceArea() {
        try {
            double radius = Double.parseDouble(txtRadius.getText().trim());
            double height = Double.parseDouble(txtHeight.getText().trim());

            if (radius <= 0 || height <= 0) {
                JOptionPane.showMessageDialog(this, "Radius and height must be positive values.", "Input Error", JOptionPane.ERROR_MESSAGE);
                return;
            }

            // Formula: Surface Area = 2 * Math.PI * r * (r + h)
            double surfaceArea = 2 * Math.PI * radius * (radius + height);

            txtResult.setText(String.format("Cylindrical Water Tank Surface Area Calculation:\n\n" +
                    "Radius (r): %.2f\n" +
                    "Height (h): %.2f\n" +
                    "Total Surface Area = %.4f square units", radius, height, surfaceArea));

        } catch (NumberFormatException ex) {
            JOptionPane.showMessageDialog(this, "Please enter valid numerical values for radius and height.", "Input Error", JOptionPane.ERROR_MESSAGE);
        }
    }

    public static void main(String[] args) {
        SwingUtilities.invokeLater(() -> new Question2().setVisible(true));
    }
}
